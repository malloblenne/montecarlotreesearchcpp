add_executable(tests src/test_main.cpp src/montecarlotreesearch_test.cpp src/games_test.cpp)
target_include_directories(tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(tests PRIVATE MonteCarloTreeSearchCPP GamesLibrary)
if(ENABLE_COVERAGE)
	target_compile_options(tests PRIVATE --coverage -O0 -g)
endif()
add_test(NAME library_tests COMMAND tests "[library]")
add_test(NAME games_tests COMMAND tests "[games]")
find_program(BASH_EXECUTABLE bash)
if(NOT BASH_EXECUTABLE)
	message(FATAL_ERROR "CLI smoke tests require Bash")
endif()
add_test(NAME cli_smoke COMMAND ${BASH_EXECUTABLE}
	${CMAKE_CURRENT_SOURCE_DIR}/cli_smoke.sh $<TARGET_FILE:Main>)
set_tests_properties(library_tests games_tests cli_smoke PROPERTIES TIMEOUT 30)
